\begin{exercise}\label{ex:la-1}
For each set below, decide whether it is a subspace of $\R^3$. Justify your answer.
\begin{enumerate}[label=(\alph*)]
  \item $W_1=\{(x,y,z)\in\R^3:x-2y+z=0\}$;
  \item $W_2=\{(x,y,z)\in\R^3:x-2y+z=1\}$;
  \item $W_3=\{(x,y,z)\in\R^3:x=y=z\}$.
\end{enumerate}
\end{exercise}

\begin{exercise}\label{ex:la-2}
Find a vector equation describing the span of $u=(1,2,-1)$ and $v=(3,1,2)$. Then determine whether $w=(4,3,1)$ lies in that span.
\end{exercise}

\begin{exercise}\label{ex:la-3}
Use row reduction to find a basis for the column space of
\[
  A=\begin{pmatrix}1&2&1&3\\2&4&3&7\\1&2&2&4\end{pmatrix}.
\]
State the dimension of that column space.
\end{exercise}

\begin{exercise}\label{ex:la-4}
Decide whether the polynomials $1+x$, $x+x^2$, and $1+2x+x^2$ are linearly independent in the vector space of real polynomials of degree at most $2$. Justify your answer from the definition.
\end{exercise}

\begin{exercise}\label{ex:la-5}
Let $u,v\in V$ be linearly independent. Prove that $u+v$ and $u-v$ are linearly independent.
\end{exercise}

\begin{exercise}\label{ex:la-6}
In a program, a collection of feature vectors is stored as the columns of an $m\times n$ real matrix $A$. Explain what it means, in the language of this lesson, for one feature column to be redundant. Then explain what the pivot columns from row reduction tell you about a nonredundant representation of the same column space.
\end{exercise}
