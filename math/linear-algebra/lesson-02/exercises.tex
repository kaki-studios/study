\begin{exercise}\label{ex:la-7}
Decide which of the following maps are linear. Justify your answer from the definition.
\begin{enumerate}[label=(\alph*)]
  \item $T_1\colon\R^2\to\R^2$, $T_1(x,y)=(2x+y,x-3y)$;
  \item $T_2\colon\R^2\to\R^2$, $T_2(x,y)=(x^2,y)$;
  \item $T_3\colon\R^2\to\R$, $T_3(x,y)=x-2y$.
\end{enumerate}
\end{exercise}

\begin{exercise}\label{ex:la-8}
Let $T\colon\R^2\to\R^3$ be the linear map with
\[
  T\begin{pmatrix}1\\1\end{pmatrix}
  =\begin{pmatrix}1\\0\\2\end{pmatrix},
  \qquad
  T\begin{pmatrix}1\\-1\end{pmatrix}
  =\begin{pmatrix}0\\1\\1\end{pmatrix}.
\]
Compute $T\begin{psmallmatrix}x\\y\end{psmallmatrix}$ for a general
$(x,y)\in\R^2$, and write down the matrix $A$ with $T(\mathbf{x})=A\mathbf{x}$.
\end{exercise}

\begin{exercise}\label{ex:la-9}
Let $T\colon\R^4\to\R^3$ be given by $T(\mathbf{x})=A\mathbf{x}$ with
\[
  A=\begin{pmatrix}1&0&2&1\\0&1&1&2\\1&1&3&3\end{pmatrix}.
\]
Find a basis for $\ker T$ and a basis for $\operatorname{im}T$, and verify
that $\dim\ker T+\dim\operatorname{im}T=4$.
\end{exercise}

\begin{exercise}\label{ex:la-10}
Let $T\colon V\to W$ be linear and injective. Prove that if
$v_1,\ldots,v_k\in V$ are linearly independent, then so are
$T(v_1),\ldots,T(v_k)\in W$.
\end{exercise}

\begin{exercise}\label{ex:la-11}
Let $V$ and $W$ be vector spaces with $\dim V=\dim W=n<\infty$, and let
$T\colon V\to W$ be linear. Prove that $T$ is injective if and only if
$T$ is surjective.
\end{exercise}

\begin{exercise}\label{ex:la-12}
A linear layer of a program maps input vectors $\mathbf{x}\in\R^n$ to
output vectors $A\mathbf{x}\in\R^m$. Explain, in the language of this
lesson, what a nonzero vector in $\ker T$ means for reconstructing the
input from the output. Then explain what the rank of $A$ tells you about
the effective number of output coordinates, and why stacking two such
linear layers (with no nonlinearity in between) can never produce more
expressive outputs than a single linear layer.
\end{exercise}
